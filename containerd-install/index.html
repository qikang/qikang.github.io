containerd - install#

安装 containerd#

# wget https://github.com/containerd/containerd/releases/download/v1.5.2/cri-containerd-cni-1.5.2-linux-amd64.tar.gz

直接解压到系统的各个子目录#

# tar -C / -xf cri-containerd-cni-1.5.2-linux-amd64.tar.gz

生成默认配置文件#

# mkdir -pv /etc/containerd

# containerd config default > /etc/containerd/config.toml

启动服务#

# systemctl daemon-reload
# systemctl status containerd
# systemctl start containerd

设置环境变量#

# echo "export PATH=$PATH:/usr/local/bin:/usr/local/sbin" | tee -a ~/.bashrc

# source ~/.bashrc

测试ctr命令#

# ctr version
