Docker - 原理#

  • docker 三大要素

    • cgroup
    • namespace
    • unionFS

  • CGroup 全称是control group，用来限定一个进程的资源使用。由Linux 内核支持，可以限制和隔离Linux进程组所使用的物理资源。比如cpu、内存、磁盘、网络IO，是Linux container技术的物理基础。

  • NameSpace 是一个资源隔离技术，主要是用来隔离进程ID，IPC，Network等系统资源 linux 共有6种 namespace 类型：

    • IPC: IPC(信号量、消息队列、共享内存等)隔离
    • Network: 网络设备、网络栈、端口等隔离
    • Mount: 文件系统挂载点隔离
    • PID: 进程编号隔离
    • User: 用户和用户组隔离
    • UTS: 主机和域名隔离 从linux内核4.6开始提供了第7种namespace: CGroup namespace主要用于cgroup根目录隔离

  • unionFS unionFS 可以把文件系统上多个目录内容联合挂载到同一个目录下，而目录的物理位置是分开的。解决了docker运行后的container挂载宿主机目录问题。

  • docker的网络模式

    DriverDescription
    bridgeThe default network driver.
    hostRemove network isolation between the container and the Docker host.
    noneCompletely isolate a container from the host and other containers.
    overlaySwarm Overlay networks connect multiple Docker daemons together.
    ipvlanConnect containers to external VLANs.
    macvlanContainers appear as devices on the host’s network.

https://docs.docker.com/engine/network/

download-docker-images.sh#

#!/bin/bash
set -x
date=`date +%H_%M_%S`
images="
platform-test/device-maintenance:dev-f5ced396
iot-device-maintenance/device-maintenance-front:v180-release
" 

harbor="harbor.test.com"  

tags=" "
for image in $images;do
  docker rmi $harbor/$image
  docker pull --platform=arm64 $harbor/$image
    tags="$harbor/$image $tags"
done

docker save $tags > tags_${date}.tar

Run Postgres#

#!/bin/bash
docker stop sonar-db
docker rm sonar-db
docker run -d \
    --name sonar-db \
    -e POSTGRES_USER=sonar-dbuser \
    -e POSTGRES_DB=sonar \
    -e POSTGRES_PASSWORD=password \
    -e PGDATA=/var/lib/postgresql/data/pgdata \
    -v /data/pgsql:/var/lib/postgresql/data \
    -p 5432:5432 \
    --restart always \
    registry.local:5000/postgres/postgres:10

Run MySQL#

#!/bin/bash
docker stop con-db
docker rm con-db
docker run -d \
  --name con-db \
  -v /data/mysql:/var/lib/mysql \
  -p 3306:3306 \
  --restart always \
  -e MYSQL_ROOT_PASSWORD=my-secret-pw \
  -e MYSQL_USER=root \
  -e MYSQL_PASSWORD=root \
  -e MYSQL_DATABASE=con \
  registry.local:5000/mysql/mysql:5.7.30

常用命令#

# 强制删除容器
docker rm -f leslie

# 查看镜像ID和镜像地址
docker images --format "table {{.ID}}\t{{.Repository}}:{{.Tag}}"

# 查看镜像地址
docker images --format "table {{.Repository}}:{{.Tag}}"

# 查看容器的name和镜像地址
docker ps -a --format="{{.Names}}\t{{.Image}}"

# 查看exited状态容器
docker ps -a -f "status=exited" --format={{.Names}}

# 查看running状态容器
docker ps -a -f "status=running" --format={{.Names}}

# 给用户添加docker权限
usermod -a -G docker ec2-user

ubuntu：/etc/default/docker
centos： /etc/sysconfig/docker

# 查看正在运行容器
docker ps

# 查看所有的容器，包括正在运行和停止的。
docker ps -a

# 查看最后两个进程
docker ps -n 2

# 获取所有容器ID
docker ps -a -q

# 查看端口映射
docker ps -l

# 只查看所有docker的name
docker ps -a --format={{.Names}}

# docker inspect
# 获取容器的运行状态
docker inspect -f='{{.State.Running}}' deamon_dave

# 获取运行中容器的PID
docker inspect -f='{{.State.Pid}}' daemon_dave

# ubuntu 后面可以加多个容器标识符
docker inspect -f='{{.State.Running}}' daemon_dave

# redis 查看redis容器的IP地址
docker inspect -f='{{.NetworkSettings.IPAddress}}' redis

# docker rm
# 删除webserver这个容器。前提是这个容器处于停止状态。
docker rm webserver

# 启动
docker start webserver

# 停止
# stop 是向容器发送SIGKILL信号
docker stop webserver

# docker attach
# 只有在容器运行，并且有可用shell的时候。才可以再次附着到已经运行的容器。

docker attach ubuntu
docker attach f1694d43a1b6
 
# docker logs
# 查看容器运行日志

docker logs deamon_dave

# 类似于tail -f实时监控deamon_dave的日志。-t为每条日志加上时间戳。
docker logs -ft deamon_dave

# 获取最后10行日志
docker logs --tail 10 deamon_dave

# 从最后一行日志开始监控
docker logs --tail 0 -f deamon_dave

# docker top
# 查看容器资源占用。

docker top deamon_dave

# docker exec
# 在容器内部运行进程
docker exec -d ubuntu touch /etc/new_file
docker exec -t -i ubuntu /bin/bash

# docker rm
# 删除不在运行的容器
docker port

# 查看容器的端口映射
# 容器内部的80/tcp端口映射到宿主机的32768随机端口上了。
docker port static_web
80/tcp -> 0.0.0.0:32768
