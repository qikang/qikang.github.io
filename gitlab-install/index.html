gitlab高可用部署方案#

  • 两台 Gitlab Rails Server 作为 Praefect nodes 同时在线
  • NFS 共享储存
  • Redis 共享
  • PortgreSQL 共享DB
  • OpenLDAP 提供统一认证
  • 官方推荐多节点部署参考地址
    • https://docs.gitlab.com/omnibus/update/#multi-node--ha-deployment 多节点部署
    • https://docs.gitlab.com/ee/administration/high_availability/gitlab.html 高可用架构推荐
    • https://docs.gitlab.com/ee/administration/reference_architectures/ PG Server、Redis、NFS部署参考

推荐配置#

  • 预计支持 2000 左右用户
  • centos 7.8.2003
  • gitlab 13.1.4
  • postgresql 11
  • nfs v4
  • redis 5.0.9
  • ldap 2.4.44
Server NameCPU核心内存SSD 磁盘大小节点数
Gitlab Rails Server16 vCPU32G>= 200G2
NFS Server4 vCPU8G>= 1T1
PostGreSQL Server4 vCPU8G>= 300G1
Redis Server2 vCPU4G>= 200G1
OpenLDAP Server2 vCPU4G>= 100G1

Gitlab服务构成#

Service Name用途
gitlab-shell用于处理Git命令和修改authorized keys列表
gitlab-workhorse轻量级的反向代理服务器
logrotate日志文件管理工具
sidekiq用于在后台执行队列任务（异步执行）
postgresql数据库
redis缓存数据库
Nginx前端web服务器
NFS文件共享存储

初始化服务器#

  • IP地址规划
    • 172.16.11.30 gitlab-1
    • 172.16.11.31 gitlab-2
    • 172.16.11.32 gitlab-db (nfs, postgresql, redis)
    • 172.16.11.33 gitlab-nginx (OpenLDAP)
  • disable firewalld selinux
  • 配置yum源，安装初始软件
    • /etc/yum.repos.d/gitlab-ce.repo
    • /etc/yum.repos.d/epel.repo
    • /etc/yum.repos.d/CentOS-Base.repo

部署 PostgreSQL Server#

添加 pgsql yum 源#

/etc/yum.repos.d/pgdg-redhat-all.repo

# wget  https://download.postgresql.org/pub/repos/yum/reporpms/EL-7-x86_64/pgdgredhat-repo-latest.noarch.rpm
# yum install -y pgdg-redhat-repo-latest.noarch.rpm
# ls -l /etc/yum.repos.d/pgdg-redhat-all.repo

安装PostgreSQL 11#

# yum install -y postgresql11-contrib postgresql11-server
# systemctl enable postgresql-11
# systemctl start postgresql-11

初始化数据库#

/usr/pgsql-11/bin/postgresql-11-setup initdb

创建用户#

# sudo -u postgres createuser -P -s -e gitlab
连续两次输入 nil 作为密码

配置文件目录#

# ls -l /var/lib/pgsql/11/data

添加授权#

# vim /var/lib/pgsql/11/data/pg_hba.conf

在最后一行添加授权策略
host    all     all             0.0.0.0/0                 md5

修改监听地址#

# vim /var/lib/pgsql/11/data/postgresql.conf
listen_addresses = '*'

重启pgsql 服务#

# systemctl restart postgresql-11

部署 NFS Server#

安装软件#

# yum install -y nfs-utils

创建目录#

# mkdir -pv /data/nfs; chmod 777 /data/nfs

添加配置文件#

# vim /etc/exports
/data/nfs 172.16.11.0/24(rw,sync,no_root_squash)

生效配置文件#

# exportfs -arv
# showmount -e
# systemctl restart nfs
# systemctl enable nfs

# mkdir -pv /data/nfs/gitlab/.ssh
# mkdir -pv /data/nfs/gitlab/gitlab-rails/uploads
# mkdir -pv /data/nfs/gitlab/gitlab-rails/shared

nfs 挂载#

在 gitlab-1 和 gitlab-2 上的 /etc/fstab 中添加 nfs 配置如下行

10.1.0.1:/var/opt/gitlab/.ssh /var/opt/gitlab/.ssh nfs4 defaults,vers=4.1,hard,rsize=1048576,wsize=1048576,noatime,nofail,lookupcache=positive 0 2

部署 Redis Server#

下载 redis 源码包#

wget http://download.redis.io/releases/redis-6.0.6.tar.gz

安装redis#

解决环境依赖
# yum -y install gcc gcc-c++ kernel-devel

升级gcc
# yum -y install centos-release-scl
# yum -y install devtoolset-9-gcc devtoolset-9-gcc-c++ devtoolset-9-binutils

永久⽣效
# echo "source /opt/rh/devtoolset-9/enable" >> /etc/profile

临时⽣效
# scl enable devtoolset-9 bash

验证版本
# gcc --version

解压源码⽂件
# tar -xf redis-6.0.6.tar.gz -C /usr/local

制作软连并部署⼆进制⽂件
# ln -sv redis-6.0.6/ redis
# cp /usr/local/redis/src/redis-server /usr/local/bin/
# cp /usr/local/redis/src/redis-cli /usr/local/bin/

创建 /etc/redis/redis.conf 配置⽂件
# mkdir /etc/redis
# cp /usr/local/redis/redis.conf /etc/redis/

创建 rdb ⽂件夹
# mkdir /var/lib/redis/

设置 /etc/redis/redis.conf ⽂件
bind 0.0.0.0
daemonize yes
dbfilename gitlab-dump.rdb
dir /var/lib/redis/

创建开机启动脚本#

# vim /usr/lib/systemd/system/redis.service
[Unit] 
Description=Redis Server 
After=network.target 
After=network-online.target 
Wants=network-online.target 
[Service] 
Type=forking 
ExecStart=/usr/local/bin/redis-server /etc/redis/redis.conf 
ExecStop=/usr/local/bin/redis-cli -h 127.0.0.1 -p 6379 shutdown 
[Install] 
WantedBy=multi-user.target

设置开机启动#

# systemctl enable redis
# systemc start redis

测试redis#

[root@RJYF-RS-01 ~]# redis-cli  
127.0.0.1:6379> info
...
查看redis info信息

# scp /etc/ssh/ssh_host_* gitlab-2:/etc/ssh/

haproxy 配置文件#

[root@RJYF-HP-01 ~]# cat /etc/haproxy/haproxy.cfg
global
    log         127.0.0.1 local2
    chroot      /var/lib/haproxy
    pidfile     /var/run/haproxy.pid
    maxconn     4000
    user        haproxy
    group       haproxy
    daemon
    stats socket /var/lib/haproxy/stats
defaults
    mode                    tcp
    log                     global
    option                  httplog
    option                  dontlognull
    option                  redispatch
    retries                 3
    timeout http-request    10s
    timeout queue           1m
    timeout connect         10s
    timeout client          1m
    timeout server          1m
    timeout http-keep-alive 10s
    timeout check           10s
    maxconn                 4000

# enable gitlab 80
frontend  gitlab_http *:80
    option              tcplog
    default_backend     backend_gitlab_http
backend backend_gitlab_http
    balance     source
    option      tcp-check
    option      httpchk GET /-/readiness?all=1
    http-check  expect status 200
    default-server inter 5s fall 3 rise 2
    server      backend_gitlab_http_server1 172.20.60.7:80 check port 80
    server      backend_gitlab_http_server2 172.20.60.8:80 check port 80

# enable gitlab 443
