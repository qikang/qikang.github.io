4. 在kernel中需要打开ipvs模块#

/etc/sysconfig/kubelet

    - ip_vs_fo \

    - ip_vs_sedip_vs_ftp \

5. 在kublet中禁用swap#

1. /etc/sysconfig/kubelet
    KUBELET_EXTRA_ARGS="--fail-swap-on=false"
2. swapoff -a
3. 在 /etc/fstab 中关闭swap
4. Enable cgroup driver /usr/lib/systemd/system/kubelet.service.d/10-kubeadm.conf

Environment=“KUBELET_CGROUP_ARGS=–cgroup-driver=systemd” cat > /etc/docker/daemon.json «EOF { “registry-mirrors”: [ “https://hub-mirror.c.163.com”, “http://hub-mirror.c.163.com”, “http://docker.mirrors.ustc.edu.cn”, “https://dockerhub.azk8s.cn” ], “insecure-registries”: [“harbor.ctfo.com:80”], “exec-opts”: [“native.cgroupdriver=systemd”], “log-driver”: “json-file”, “log-opts”: { “max-size”: “100m” }, “storage-driver”: “overlay2”, “storage-opts”: [ “overlay2.override_kernel_check=true” ] } EOF
