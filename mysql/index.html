my.cnf 配置文件#

mysql my.cnf加载顺序#

/etc/my.cnf
/etc/mysql/my.cnf
/path/to/file --defaults-extra-file=/path/to/file
~/.my.cnf

设置家目录下的my.cnf#

vim ~/.my.cnf
[client]
socket=/var/lib/mysql/mysql.sock
host=localhost
password=qikang

mysql cli#

-D 连接那个数据库
-H 以html格式输出
-X 以xml格式输出
-e 执行的MySQL语句
# mysql -uroot -p -D mysql -H -e "select user,host,password from user;" > a.html

--compress #以压缩的格式连接mysql
# mysql -uroot -p --compress

# 测试连通性
[root@rsync1 ~]# mysqladmin -uroot -p ping

# 创建xikang库
[root@rsync1 ~]# mysqladmin -uroot -p create xikang

# 查看创建的qikang库
[root@rsync1 ~]# mysql -uroot -p -e 'show databases;'

# 查看状态
[root@rsync1 ~]# mysqladmin -uroot -p status
Uptime: 2982 Threads: 2 Questions: 126 Slow queries: 0 Opens: 87 Flush tables: 1 Open tables: 80 Queries per second avg: 0.042

# 1秒刷新一次，显示5次
[root@rsync1 ~]# mysqladmin -uroot -p status --sleep 1 --count 5

# 查看MySQL状态的扩展信息，相当于mysql> SHOW GLOBAL STATUS;
[root@rsync1 ~]# mysqladmin -uroot -p extended-status

# 清除dns缓存和主机缓存表记录
[root@rsync1 ~]# mysqladmin -uroot -p flush-hosts

# 刷新权限
[root@rsync1 ~]# mysqladmin -uroot -p flush-privileges

# 刷新日志，做日志回滚
[root@rsync1 ~]# mysqladmin -uroot -p flush-logs

# 刷新状态，重置状态变量
[root@rsync1 ~]# mysqladmin -uroot -p flush-status

# 清空线程缓存池
[root@rsync1 ~]# mysqladmin -uroot -p flush-threads

# 查看mysql线程
[root@rsync1 ~]# mysqladmin -uroot -p processlist
refresh 相当与flush-privileges和flush-logs
start-slave 启动从服务器进程
stop-slave 挂壁从服务器进程
variables 输出mysqld个 服务器变量

> SHOW VARIABLES LIKE  '%binlog%';

# 查找id最大一条记录
> SELECT id from mr2_reservations ORDER BY id DESC LIMIT 1\G;

# 增加一个字段，默认为空
> alter table table_name add COLUMN new1 VARCHAR(20) DEFAULT NULL;

# 增加一个字段，默认不能为空
> alter table table_name add COLUMN new2 VARCHAR(20) NOT NULL;

# 删除一个字段
> alter table table_name DROP COLUMN new2;

# 修改一个字段的类型
> alter table table_name MODIFY new1 VARCHAR(10);

# 修改一个字段的名称，此时一定要重新指定该字段的类型
> alter table table_name CHANGE new1 new4 int;

# 获取表结构字段的名字
> select column_name from information_schema.columns where table_schema ='chopeReserveDB' and table_name = 'user_info';

# 查看所有mysql进程
> show full processlist\G;

# 分页取数据,从2开始偏移7。也就是 2 - 9 行的数据
> select * from table_name limit 2, 7;
