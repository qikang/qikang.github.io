skopeo 下载镜像工具#

skopeo version 1.20

同步互联网镜像到znlkhub#

skopeo copy -a --insecure-policy --dest-tls-verify=false --src-tls-verify=false \
  docker://hub.zentao.net/app/zentao:21.7.5 \
  docker://harbor.internal.com/zentao/zentao:21.7.5

下载镜像,并打tag#

skopeo copy --override-arch arm64 --src-tls-verify=false \
  docker://harbor.internal.com/devops-public/tdengine:3.2.2.0 \
  docker-archive:harbor.internal.com_devops-public_tdengine_3.2.2.0.tar:harbor.internal.com/devops-public/tdengine:3.2.2.0

查看远程镜像#

skopeo inspect --tls-verify=false docker://127.0.0.1:5000/devops-public/busybox:1.32
skopeo inspect --tls-verify=false docker://172.19.63.45:5000/devops-public/busybox:1.32

查看远程镜像支持的架构#

skopeo inspect --raw docker://harbor.internal.com/devops-public/busybox:1.32
skopeo inspect --tls-verify=false --raw docker://172.19.63.45:5000/devops-public/busybox:1.32 | jq
skopeo inspect --tls-verify=false --raw docker://127.0.0.1:5000/znlk-edge/app-manage:v1.3.0.0 | jq

下载远程镜像到本地镜像仓库#

skopeo copy \
  docker://harbor.internal.com/devops-public/busybox:1.32 \
  docker://127.0.0.1:5000/devops-public/busybox:1.32 \
  --dest-tls-verify=false
skopeo copy --insecure-policy --dest-tls-verify=false --src-tls-verify=false

下载离线镜像#

HTTP_PROXY=http://192.168.56.1:7897/ \
HTTPS_PROXY=http://192.168.56.1:7897/ \
NO_PROXY=localhost,127.0.0.0/8,::1,172.20.0.0/16,harbor.internal.com \
skopeo copy --override-arch arm64 --insecure-policy \
  docker://docker.io/library/nginx:1.25.3 \
  oci-archive:arm64.tar

下载离线镜像#

HTTP_PROXY=http://192.168.56.1:7897/ \
HTTPS_PROXY=http://192.168.56.1:7897/ \
NO_PROXY=localhost,127.0.0.0/8,::1,172.20.0.0/16,harbor.internal.com \
skopeo copy --override-arch amd64 --insecure-policy \
  docker://docker.io/library/nginx:1.25.3 \
  oci-archive:amd64.tar

上传 amd64 离线镜像#

skopeo copy --insecure-policy \
  oci-archive:amd64.tar \
  docker://myregistry.local:5000/nginx:1.25.3-amd64

上传 arm64 离线镜像#

skopeo copy --insecure-policy \
  oci-archive:arm64.tar \
  docker://myregistry.local:5000/nginx:1.25.3-arm64

创建一个空的 oci layout#

mkdir nginx-oci
skopeo copy oci-archive:nginx-amd64.tar oci:nginx-oci:amd64
skopeo copy oci-archive:nginx-arm64.tar oci:nginx-oci:arm64

合并#

skopeo copy --all oci:nginx-oci:latest docker://myregistry.local:5000/nginx:1.25.3
